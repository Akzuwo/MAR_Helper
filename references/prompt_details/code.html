import React from 'react';

const UserProfileCard = ({ user }) => {
  return (
    <div className="bg-white border border-gray-200 rounded-xl p-6 max-w-sm w-full mx-auto shadow-sm">
      <div className="flex items-center gap-4 mb-4">
        <img

          <p className="text-sm text-gray-500">{user.role}</p>
        </div>
      </div>
      
      <p className="text-gray-600 text-sm mb-6 line-clamp-3">
        {user.bio}
      </p>
      
      <div className="flex gap-3">
        <button className="flex-1 bg-indigo-600 hover:bg-indigo-700 text-white text-sm font-medium py-2 px-4 rounded-lg transition-colors">
          Follow
        </button>
        <button className="flex-1 bg-white hover:bg-gray-50 text-gray-900 text-sm font-medium py-2 px-4 rounded-lg border border-gray-200 transition-colors">
          Message

    </div>
  );

export default UserProfileCard;
